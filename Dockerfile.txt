# Build stage
FROM node:20-alpine AS builder

WORKDIR /app

# Copy package manifests and install dependencies
COPY package*.json ./
RUN npm install --legacy-peer-deps

# Copy source files and build production bundle
COPY . .
RUN npm run build

# Production serve stage (Hugging Face Spaces runs as user 1000 on port 7860)
FROM node:20-alpine

WORKDIR /app

# Install lightweight SPA web server
RUN npm install -g serve

# Create non-root user matching Hugging Face standard permissions (UID 1000)
RUN adduser -D -u 1000 appuser

# Copy compiled static distribution from builder
COPY --from=builder /app/dist ./dist
RUN chown -R appuser:appuser /app

USER appuser

# Hugging Face Spaces port
EXPOSE 7860

# Serve SPA on port 7860
CMD ["serve", "-s", "dist", "-l", "7860"]
